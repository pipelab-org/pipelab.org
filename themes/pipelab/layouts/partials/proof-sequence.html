A scene joins the deck only when its manifest carries a block receipt with every rendered field and a payload shape; the build fails rather than rendering a plausible-looking sequence from incomplete evidence. Keep this template math-free: the pinned Hugo in .hugo-version rejects `mul` on YAML/JSON values, so all timing lives in the stylesheet. */}} {{- $galleryPath := "static/playground-replay/gallery.json" -}} {{- $gallery := transform.Unmarshal (readFile $galleryPath) -}} {{- $allowPath := "static/playground-replay/allowed-safe-read/manifest.json" -}} {{- $allow := transform.Unmarshal (readFile $allowPath) -}} {{- if not (isset $allow "destination_class") -}} {{- errorf "proof-sequence.html: %s is missing destination_class" $allowPath -}} {{- else if not $allow.destination_class -}} {{- errorf "proof-sequence.html: %s has an empty destination_class" $allowPath -}} {{- end -}} {{/* The anchor scene renders first: the credential-in-URL block reads best cold. The rest of the deck follows gallery order. */}} {{- $anchorID := "secret-exfil-url-blocked" -}} {{- $orderedIDs := slice $anchorID -}} {{- range $gallery.scenarios -}} {{- if ne .id $anchorID -}}{{- $orderedIDs = $orderedIDs | append .id -}}{{- end -}} {{- end -}} {{/* Scene IDs become build-time file paths and on-page link targets, so they are constrained to a strict slug here rather than trusted. ids become file paths and link targets" $sid -}} {{- end -}} {{- $mPath := printf "static/playground-replay/%s/manifest.json" $sid -}} {{- if fileExists $mPath -}} {{- $m := transform.Unmarshal (readFile $mPath) -}} {{- if eq $m.decisive_verdict "block" -}} {{- $blockReceipt := dict -}} {{- $dupBlock := false -}} {{- range $m.receipts -}} {{- if eq .verdict "block" -}} {{- if $blockReceipt -}}{{- $dupBlock = true -}}{{- end -}} {{- $blockReceipt = . -}} {{- end -}} {{- end -}} {{- if $dupBlock -}} {{- errorf "proof-sequence.html: %s has more than one block receipt; the deck needs one decisive receipt per scene" $mPath -}} {{- end -}} {{- $shapeOK := false -}} {{- $payloadShape := "" -}} {{- if and (isset $m "redacted_shape") $m.redacted_shape -}} {{- $parts := split $m.redacted_shape " → " -}} {{- if eq (len $parts) 2 -}} {{- $payloadShape = index $parts 0 -}} {{- if and $payloadShape (index $parts 1) -}}{{- $shapeOK = true -}}{{- end -}} {{- end -}} {{- end -}} {{- $fieldsOK := true -}} {{- range $f := slice "without" "with" "pipelock_version" "captured_at" "policy_hash" "signer_key" "destination_class" "verifier_command" "title" -}} {{- if or (not (isset $m $f)) (not (index $m $f)) -}}{{- $fieldsOK = false -}}{{- end -}} {{- end -}} {{- if $blockReceipt -}} {{- if not (isset $blockReceipt "chain_seq") -}}{{- $fieldsOK = false -}}{{- end -}} {{- range $f := slice "verdict" "layer" "pattern" "action_id" "target_redacted" "transport" -}} {{- if or (not (isset $blockReceipt $f)) (not (index $blockReceipt $f)) -}}{{- $fieldsOK = false -}}{{- end -}} {{- end -}} {{- else -}} {{- $fieldsOK = false -}} {{- end -}} {{/* The manifest must at least name the pinned lab signer that shipped with the gallery, or the build fails. Tracked as follow-up. */}} {{- if and $fieldsOK (ne $m.signer_key $signerKey) -}} {{- errorf "proof-sequence.html: %s claims signer %q, which is not the pinned gallery signer" $mPath $m.signer_key -}} {{- end -}} {{- if and $shapeOK $fieldsOK -}} {{- $target := urls.Parse $blockReceipt.target_redacted -}} {{- $host := "" -}} {{- with $target -}}{{- $host = .Hostname -}}{{- end -}} {{- if not $host -}} {{- errorf "proof-sequence.html: target_redacted in %s must be an absolute URL with a hostname" $mPath -}} {{- end -}} {{- $scenes = $scenes | append (dict "id" $sid "title" $m.title "without" $m.without "with" $m.with "shape" $payloadShape "pattern" $blockReceipt.pattern "layer" $blockReceipt.layer "verdict" $blockReceipt.verdict "action_id" $blockReceipt.action_id "chain_seq" $blockReceipt.chain_seq "transport" $blockReceipt.transport "target" $blockReceipt.target_redacted "host" $host "policy_short" (substr $m.policy_hash 0 22) "signer_head" (substr $m.signer_key 0 8) "signer_tail" (substr $m.signer_key -8) "version" $m.pipelock_version "date" (dateFormat "Jan 2, 2006" $m.captured_at) "replay_url" (printf "/playground/replay/%s/" $sid) "manifest_url" (printf "/playground-replay/%s/manifest.json" $sid) "verifier_command" $m.verifier_command ) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{/* Six scenes qualify today. A regression that silently shrinks the deck is a content fail-open, so the floor is enforced; raising the deck is fine. */}} {{- if lt (len $scenes) 4 -}} {{- errorf "proof-sequence.html: only %d scenes qualified for the deck (expected at least 4); a manifest lost a required field" (len $scenes) -}} {{- end -}} {{- $first := index $scenes 0 -}} {{- if ne $first.id $anchorID -}} {{- errorf "proof-sequence.html: the anchor scene %q did not qualify for the deck" $anchorID -}} {{- end -}}
