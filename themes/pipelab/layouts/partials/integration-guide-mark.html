{{- /* Reuse the homepage's guide mapping and official assets. A guide the orbit links to gets that entry's mark. Any other page can name an entry with `title_brand: ""` (orbit or tile rows in data/integrations.yaml), so its logo and colour still come from that one file; the reserved name "Pipelock" is the product mark. */ -}} {{- $page := . -}} {{- $colorPattern := `^#[0-9A-Fa-f]{6}$` -}} {{- $match := dict -}} {{- with $page.Params.title_brand -}} {{- $want := printf "%v" . -}} {{- if eq $want "Pipelock" -}} {{- $match = dict "name" "Pipelock" "color" "#00e5a0" "brand" "pipelock-mark" -}} {{- else -}} {{- range site.Data.integrations.orbit -}} {{- if and (not $match.name) (eq .name $want) (ne .kind "protocol") -}}{{- $match = . -}}{{- end -}} {{- end -}} {{- range site.Data.integrations.rows -}} {{- range .items -}} {{- if and (not $match.name) (eq .name $want) -}}{{- $match = . -}}{{- end -}} {{- end -}} {{- end -}} {{- if not $match.name -}} {{- errorf "%s: title_brand %q names no integration in data/integrations.yaml" $page.File.Path $want -}} {{- end -}} {{- end -}} {{- else -}} {{- range site.Data.integrations.orbit -}} {{- $guide := site.GetPage .href -}} {{- if and (not $match.name) (ne .kind "protocol") $guide (eq $guide.RelPermalink $page.RelPermalink) -}}{{- $match = . -}}{{- end -}} {{- end -}} {{- end -}} {{- with $match.name -}} {{- if ne (len (findRE $colorPattern (printf "%v" $match.color))) 1 -}} {{- errorf "integration guide %q: invalid brand color %q" $match.name $match.color -}} {{- end -}} {{- $hasImg := partial "integration-mark.html" (dict "entry" $match "context" "integration guide" "required" true) -}} {{- if $match.brand -}} {{- partial (printf "brands/%s.svg" $match.brand) . -}} {{- else if $hasImg -}} {{- end -}} {{- end -}}
