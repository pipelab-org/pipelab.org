{{- /* Reuse the homepage's guide mapping and official assets. A guide the orbit links to gets that entry's mark. Any other page can set `title_brand` to an entry name (orbit or tile rows in data/integrations.yaml) or to a brand mark one of those entries uses, so its logo and colour still come from that one file; the reserved name "Pipelock" is the product mark. */ -}} {{- $page := . -}} {{- $colorPattern := `^#[0-9A-Fa-f]{6}$` -}} {{- $match := dict -}} {{- with $page.Params.title_brand -}} {{- $want := printf "%v" . -}} {{- if eq $want "Pipelock" -}} {{- $match = dict "name" "Pipelock" "color" "#00e5a0" "brand" "pipelock-mark" -}} {{- else -}} {{- range site.Data.integrations.orbit -}} {{- if and (not $match.name) (eq .name $want) (ne .kind "protocol") -}}{{- $match = . -}}{{- end -}} {{- end -}} {{- range site.Data.integrations.rows -}} {{- range .items -}} {{- if and (not $match.name) (eq .name $want) -}}{{- $match = . -}}{{- end -}} {{- end -}} {{- end -}} {{- /* A page about another product from the same vendor names the brand mark itself (`title_brand: "cloudflare"`), so it doesn't borrow a different product's entry; colour comes from the first entry using it. */ -}} {{- range site.Data.integrations.orbit -}} {{- if and (not $match.name) (eq (printf "%v" .brand) $want) -}}{{- $match = dict "name" $want "color" .color "brand" .brand -}}{{- end -}} {{- end -}} {{- range site.Data.integrations.rows -}} {{- range .items -}} {{- if and (not $match.name) (eq (printf "%v" .brand) $want) -}}{{- $match = dict "name" $want "color" .color "brand" .brand -}}{{- end -}} {{- end -}} {{- end -}} {{- if not $match.name -}} {{- errorf "%s: title_brand %q names no integration or brand mark in data/integrations.yaml" $page.File.Path $want -}} {{- end -}} {{- end -}} {{- else -}} {{- range site.Data.integrations.orbit -}} {{- $guide := site.GetPage .href -}} {{- if and (not $match.name) (ne .kind "protocol") $guide (eq $guide.RelPermalink $page.RelPermalink) -}}{{- $match = . -}}{{- end -}} {{- end -}} {{- end -}} {{- with $match.name -}} {{- if ne (len (findRE $colorPattern (printf "%v" $match.color))) 1 -}} {{- errorf "integration guide %q: invalid brand color %q" $match.name $match.color -}} {{- end -}} {{- $hasImg := partial "integration-mark.html" (dict "entry" $match "context" "integration guide" "required" true) -}} {{- if $match.brand -}} {{- partial (printf "brands/%s.svg" $match.brand) . -}} {{- else if $hasImg -}} {{- end -}} {{- end -}}
