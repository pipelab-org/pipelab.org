{{- /* Renders a full install code block from data/install.yaml. Methods appear in the data file's order, each preceded by a `# Label` comment, so Go always leads and Homebrew always trails no matter how the caller lists the ids. Usage: {{< install-block >}} all methods {{< install-block "go,homebrew" >}} a subset, still in data order Renders through Hugo's highlighter so it matches hand-written bash blocks. */ -}} {{- $only := "" -}} {{- with .Get 0 -}} {{- $only = . -}} {{- end -}} {{- $args := dict -}} {{- if $only -}} {{- $ids := slice -}} {{- range (split $only ",") -}} {{- with (trim . " ") -}} {{- $ids = $ids | append . -}} {{- end -}} {{- end -}} {{- $args = dict "only" $ids -}} {{- end -}} {{- $methods := partial "install/methods.html" $args -}} {{/* One group per method, separated by a blank line, matching the hand-written blocks these replaced. */}} {{- $groups := slice -}} {{- range $methods -}} {{- $groups = $groups | append (printf "# %s\n%s" .label .command) -}} {{- end -}} {{- highlight (delimit $groups "\n\n") "bash" "" -}}
